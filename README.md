## `> cat RBAC_UI.md`

```tsx
// Role-Based UI — Washington Forum Frontend

// Hook de permissão
const { hasPermission, hasRole } = usePermission()

// Componente protegido por role
<ProtectedRoute role="admin">
  <AdminDashboard />
</ProtectedRoute>

// Renderização condicional por permissão
{hasPermission('factions:manage') && (
  <Button onClick={openFactionForm}>Criar Faction</Button>
)}

// Renderização condicional por role
{hasRole('staff') && (
  <ApplicationReviewPanel applications={pending} />
)}

// Guard de rota completo
function ProtectedRoute({ children, role }: Props) {
  const { user, isAuthenticated } = useAuth()

  if (!isAuthenticated) return <Navigate to="/login" />
  if (role && !user?.roles.includes(role)) return <Navigate to="/" />

  return children
}
```

## `> cat API_CLIENT.md`

```typescript
// src/api/client.ts — Axios com interceptors JWT automáticos

import axios from 'axios'
import { getToken, setToken, clearTokens } from '@/utils/token'

const api = axios.create({
  baseURL: import.meta.env.VITE_API_URL,
  timeout: 10_000,
})

// REQUEST — injeta Bearer token automaticamente
api.interceptors.request.use((config) => {
  const token = getToken('access')
  if (token) config.headers.Authorization = `Bearer ${token}`
  return config
})

// RESPONSE — refresh automático em 401
api.interceptors.response.use(
  (res) => res,
  async (error) => {
    const original = error.config
    if (error.response?.status === 401 && !original._retry) {
      original._retry = true
      const refreshToken = getToken('refresh')
      const { data } = await api.post('/auth/refresh', { refreshToken })
      setToken('access', data.accessToken)
      original.headers.Authorization = `Bearer ${data.accessToken}`
      return api(original)
    }
    clearTokens()
    window.location.href = '/login'
    return Promise.reject(error)
  }
)

export default api
```
